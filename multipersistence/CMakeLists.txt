cmake_minimum_required(VERSION 3.16)
project(multipersistence VERSION 0.1.0 LANGUAGES CXX)
add_executable(multipersistence src/main.cpp)
target_compile_features(multipersistence PRIVATE cxx_std_17)
set_target_properties(multipersistence PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(multipersistence PRIVATE /W4 /permissive-)
else()
  target_compile_options(multipersistence PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(quotient_test tests/quotient.cpp)
  target_compile_features(quotient_test PRIVATE cxx_std_17)
  target_include_directories(quotient_test PRIVATE src)
  add_test(NAME quotient_projection COMMAND quotient_test)
  add_executable(target_cache_test tests/target_cache.cpp)
  target_compile_features(target_cache_test PRIVATE cxx_std_17)
  target_include_directories(target_cache_test PRIVATE src)
  add_test(NAME target_cache_reuse COMMAND target_cache_test "${CMAKE_CURRENT_SOURCE_DIR}/examples/slices-a.txt")
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../english")
      add_test(NAME manuscript_references COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/manuscript_references.py")
    endif()
    add_test(NAME mathematical_regressions COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/regression.py" "$<TARGET_FILE:multipersistence>")
    add_test(NAME indexed_intersections COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/index_regression.py" "$<TARGET_FILE:multipersistence>")
    add_test(NAME lazy_and_tabulated COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/query_modes.py" "$<TARGET_FILE:multipersistence>")
  endif()
endif()
